    def get_driver_version(self, driver_path):
        try:
            result = subprocess.run([driver_path, "--version"], capture_output=True, text=True)
            version = result.stdout.strip()
            match = re.search(r"(\d+\.\d+\.\d+\.\d+)", version)
            return match.group(1) if match else version
        except Exception as e:
            return f"Erro ao verificar versão do driver: {str(e)}"

    def get_browser_version_from_registry(self, browser_name):
        try:
            if browser_name.lower() == "chrome":
                key_path = r"Software\Google\Chrome\BLBeacon"
                with winreg.OpenKey(winreg.HKEY_CURRENT_USER, key_path) as key:
                    version, _ = winreg.QueryValueEx(key, "version")
                    match = re.search(r"(\d+\.\d+\.\d+\.\d+)", version)
                    return match.group(1) if match else version
            elif browser_name.lower() == "edge":
                key_path = r"Software\Microsoft\Edge\BLBeacon"
                with winreg.OpenKey(winreg.HKEY_CURRENT_USER, key_path) as key:
                    version, _ = winreg.QueryValueEx(key, "version")
                    match = re.search(r"(\d+\.\d+\.\d+\.\d+)", version)
                    return match.group(1) if match else version
            return f"{browser_name} não encontrado no registro"
        except Exception as e:
            return f"Erro ao verificar versão no registro: {str(e)}"

    def get_browser_version(self, browser_name):
        try:
            if browser_name.lower() == "chrome":
                paths = [
                    r"C:\Program Files\Google\Chrome\Application\chrome.exe",
                    r"C:\Program Files (x86)\Google\Chrome\Application\chrome.exe",
                    r"C:\Users\efonseca\AppData\Local\Google\Chrome\Application\chrome.exe"
                ]
                for path in paths:
                    if os.path.exists(path):
                        self.log_message(f"Tentando verificar versão do Chrome em: {path}")

                        version = result.stdout.strip()
                        match = re.search(r"(\d+\.\d+\.\d+\.\d+)", version)
                        if match:
                            return match.group(1)

                return self.get_browser_version_from_registry("chrome")
            elif browser_name.lower() == "edge":
                paths = [
                    r"C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe",
                    r"C:\Program Files\Microsoft\Edge\Application\msedge.exe",
                    r"C:\Users\efonseca\AppData\Local\Microsoft\Edge\Application\msedge.exe"
                ]
                for path in paths:
                    if os.path.exists(path):
                        self.log_message(f"Tentando verificar versão do Edge em: {path}")
                        result = subprocess.run([path, "--version"], capture_output=True, text=True, shell=True)
                        version = result.stdout.strip()
                        match = re.search(r"(\d+\.\d+\.\d+\.\d+)", version)
                        if match:
                            return match.group(1)
                        self.log_message(f"Saída inesperada ao verificar versão do Edge: {version}")
                self.log_message("Edge não encontrado nos caminhos padrão. Tentando verificar no registro.")
                return self.get_browser_version_from_registry("edge")
            return f"{browser_name} não encontrado"
        except Exception as e:
            self.log_message(f"Erro ao verificar versão do navegador {browser_name}: {str(e)}")
            return f"Erro: {str(e)}"

    def close_browser_instances(self, browser_name):
        try:
            if browser_name.lower() == "chrome":
                subprocess.run(["taskkill", "/IM", "chrome.exe", "/F"], capture_output=True, text=True)
                self.log_message("Sessões abertas do Chrome foram fechadas.")
            elif browser_name.lower() == "edge":
                subprocess.run(["taskkill", "/IM", "msedge.exe", "/F"], capture_output=True, text=True)
                self.log_message("Sessões abertas do Edge foram fechadas.")
        except subprocess.CalledProcessError as e:
            self.log_message(f"Erro ao fechar instâncias do {browser_name}: {str(e)}")

    def setup_driver(self, browser_choice):
        try:
            self.log_message(f"Iniciando configuração do driver para {browser_choice.capitalize()}...")
            self.close_browser_instances(browser_choice)

            if browser_choice.lower() == "chrome":
                driver_path = os.path.join(os.getcwd(), "webdrivers", "chromedriver", "chromedriver.exe")
                self.log_message(f"Verificando ChromeDriver em: {driver_path}")
                if not os.path.exists(driver_path):
                    self.log_message(f"Erro: ChromeDriver não encontrado em {driver_path}. Verifique se o arquivo existe na pasta webdrivers\\chromedriver.")
                    raise FileNotFoundError(f"ChromeDriver não encontrado em {driver_path}")

                # Verifica versões
                driver_version = self.get_driver_version(driver_path)
                self.log_message(f"Versão do ChromeDriver: {driver_version}")
                browser_version = self.get_browser_version("chrome")
                self.log_message(f"Versão do Chrome detectada: {browser_version}")

                # Configura o Chrome
                chrome_options = ChromeOptions()
                chrome_options.add_argument("--disable-extensions")
                chrome_options.add_argument("--disable-background-networking")
                chrome_options.add_argument("--no-sandbox")
                chrome_options.add_argument("--disable-gpu")
                chrome_options.add_argument("--start-maximized")

                service = ChromeService(driver_path)
                self.log_message("Tentando iniciar o Chrome via Selenium...")
                self.driver = webdriver.Chrome(service=service, options=chrome_options)
                self.log_message("Chrome iniciado com sucesso pelo Selenium.")

            elif browser_choice.lower() == "edge":
                driver_path = os.path.join(os.getcwd(), "webdrivers", "edgedriver", "msedgedriver.exe")
                self.log_message(f"Verificando EdgeDriver em: {driver_path}")
                if not os.path.exists(driver_path):
                    self.log_message(f"Erro: EdgeDriver não encontrado em {driver_path}. Verifique se o arquivo existe na pasta webdrivers\\edgedriver.")
                    raise FileNotFoundError(f"EdgeDriver não encontrado em {driver_path}")

                # Verifica versões
                driver_version = self.get_driver_version(driver_path)
                self.log_message(f"Versão do EdgeDriver: {driver_version}")
                browser_version = self.get_browser_version("edge")
                self.log_message(f"Versão do Edge detectada: {browser_version}")

                # Configura o Edge
                edge_options = EdgeOptions()
                edge_options.add_argument("--disable-extensions")
                edge_options.add_argument("--disable-background-networking")
                edge_options.add_argument("--no-sandbox")
                edge_options.add_argument("--disable-gpu")
                edge_options.add_argument("--start-maximized")

                service = EdgeService(driver_path)
                self.log_message("Tentando iniciar o Edge via Selenium...")
                self.driver = webdriver.Edge(service=service, options=edge_options)
                self.log_message("Edge iniciado com sucesso pelo Selenium.")

            else:
                raise ValueError("Erro: Navegador inválido selecionado. Escolha 'Chrome' ou 'Edge'.")
        except SessionNotCreatedException as e:
            self.log_message(f"Erro: Não foi possível iniciar o navegador via Selenium. Verifique a compatibilidade entre o navegador e o driver. Detalhes: {str(e)}")
            self.log_message(f"Ação necessária: Atualize o {browser_choice} para a versão 136.x em https://www.google.com/chrome/ ou https://www.microsoft.com/edge")
            self.log_message(f"Baixe o driver correspondente em https://chromedriver.chromium.org/downloads ou https://developer.microsoft.com/en-us/microsoft-edge/tools/webdriver/")
            self.status_label.config(text="Erro no driver", bootstyle="danger")
            raise
        except WebDriverException as e:
            self.log_message(f"Erro no WebDriver: {str(e)}")
            self.log_message(f"Verifique se o {browser_choice} está instalado e se o driver em {driver_path} é compatível.")
            self.status_label.config(text="Erro no driver", bootstyle="danger")
            raise
        except Exception as e:
            self.log_message(f"Erro ao configurar o driver: {str(e)}")
            self.status_label.config(text="Erro no driver", bootstyle="danger")
            raise

    def choose_browser(self):
        response = Messagebox.show_question(
            title="Escolher Navegador",
            message="Qual navegador deseja usar para o mapeamento?",
            buttons=["Chrome:primary", "Edge:secondary", "Cancelar:danger"]
        )
        if response == "Chrome":
            return "chrome"
        elif response == "Edge":
            return "edge"
        else:
            return None

    def capture_user_actions(self, url, flow_name, browser_choice):
        try:
            self.log_message(f"Carregando URL {url} no {browser_choice.capitalize()}...")
            self.driver.get(url)
            self.actions_log = []

            # Injetar script JS para capturar eventos
            self.log_message("Injetando script de captura de ações...")
            self.driver.execute_script("""
                window.actionLog = [];
                document.addEventListener('click', function(e) {
                    let selector = e.target.id ? '#' + e.target.id : 
                                  e.target.className ? '.' + e.target.className.split(' ').join('.') :
                                  e.target.tagName.toLowerCase();
                    window.actionLog.push({
                        action: 'click',
                        element: selector,
                        value: null
                    });
                });
                document.addEventListener('input', function(e) {
                    if (e.target.value) {
                        let selector = e.target.id ? '#' + e.target.id : 
                                      e.target.className ? '.' + e.target.className.split(' ').join('.') :
                                      e.target.tagName.toLowerCase();
                        window.actionLog.push({
                            action: 'input',
                            element: selector,
                            value: e.target.value
                        });
                    }
                });
            """)
            self.log_message("Script de captura injetado com sucesso. Realize o fluxo '{flow_name}' no navegador. Pressione 'q' para encerrar e visualizar o log de ações.")

            # Aguarda até o usuário pressionar 'q'
            while not keyboard.is_pressed('q'):
                time.sleep(0.1)

            # Recupera o log de ações
            self.actions_log = self.driver.execute_script("return window.actionLog;")
            self.log_message(f"Mapeamento encerrado pelo usuário. Ações capturadas: {len(self.actions_log)}")
            for action in self.actions_log:
                self.log_message(f"Ação capturada: {action}")
            self.driver.quit()
            self.driver = None
            self.status_label.config(text="Mapeamento concluído", bootstyle="success")
            self.generate_button.config(state="normal")
            self.start_button.config(state="normal")
        except Exception as e:
            self.log_message(f"Erro durante o mapeamento: {str(e)}")
            self.status_label.config(text="Erro no mapeamento", bootstyle="danger")
            if self.driver:
                self.driver.quit()
                self.driver = None
            raise

    def start_mapping(self):
        url = self.url_entry.get()
        flow_name = self.flow_entry.get()
        if not url or not flow_name:
            self.log_message("Erro: Preencha a URL e o nome do fluxo.")
            return

        # Pergunta qual navegador usar
        browser_choice = self.choose_browser()
        if not browser_choice:
            self.log_message("Mapeamento cancelado pelo usuário.")
            return

        self.status_label.config(text="Mapeamento em andamento...", bootstyle="warning")
        self.start_button.config(state="disabled")
        try:
            self.setup_driver(browser_choice)
            # Executa o mapeamento em uma thread separada para não travar a GUI
            threading.Thread(target=self.capture_user_actions, args=(url, flow_name, browser_choice), daemon=True).start()
        except Exception as e:
            self.log_message(f"Erro ao iniciar o mapeamento: {str(e)}")
            self.start_button.config(state="normal")
            self.status_label.config(text="Erro no mapeamento", bootstyle="danger")

    def generate_cypress_test(self, actions_log, flow_name, api_key):
        prompt = f"""
        Com base no seguinte log de ações realizadas em um sistema, gere um teste automatizado em Cypress para replicar o fluxo '{flow_name}'. Use seletores robustos, boas práticas e inclua verificações. Log: {json.dumps(actions_log, indent=2)}.
        Exemplo de saída:
        describe('{flow_name}', () => {{
          it('Deve executar o fluxo {flow_name}', () => {{
            cy.visit('https://exemplo.com');
            cy.get('#loginButton').click();
            cy.get('#username').type('teste123');
            cy.get('#password').type('senha123');
            cy.get('button[type="submit"]').click();
            cy.url().should('include', '/dashboard');
          }}));
        }});
        """
        
        commands = []
        for action in actions_log:
            if action['action'] == 'input':
                commands.append(f"cy.get('{action['element']}').type('{action['value']}');")
            else:
                commands.append(f"cy.get('{action['element']}').click();")
        
        self.cypress_code = f"""
describe('{flow_name}', () => {{
  it('Deve executar o fluxo {flow_name}', () => {{
    cy.visit('{self.url_entry.get()}');
    {''.join(commands)}
    cy.url().should('include', '/success');
  }}));
}});
"""
        self.log_message("Teste Cypress gerado com sucesso!")
        self.status_label.config(text="Teste gerado", bootstyle="success")
        self.save_button.config(state="normal")
        return self.cypress_code

    def generate_test(self):
        api_key = self.api_key_entry.get()
        flow_name = self.flow_entry.get()
        if not api_key:
            self.log_message("Erro: Preencha a chave da API.")
            return

        self.status_label.config(text="Gerando teste...", bootstyle="warning")
        self.generate_button.config(state="disabled")
        self.cypress_code = self.generate_cypress_test(self.actions_log, flow_name, api_key)

    def save_test(self):
        flow_name = self.flow_entry.get()
        if not self.cypress_code:
            self.log_message("Erro: Nenhum teste gerado para salvar.")
            return

        filename = f"{flow_name.lower().replace(' ', '_')}.cy.js"
        with open(filename, "w") as f:
            f.write(self.cypress_code)
        self.log_message(f"Teste salvo em {filename}")
        self.status_label.config(text="Teste salvo", bootstyle="success")
        self.save_button.config(state="disabled")
        self.start_button.config(state="normal")

if __name__ == "__main__":
    root = Window()
    app = TestGeneratorApp(root)
    root.mainloop()
