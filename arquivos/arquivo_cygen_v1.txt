    def process_ui_queue(self):
        """Processa a fila de atualizações da UI de forma segura na thread principal"""
        try:
            while not self.ui_queue.empty():
                task = self.ui_queue.get_nowait()
                if task:
                    task() # Executa a função lambda ou callable
                self.ui_queue.task_done()
        except queue.Empty:
            pass # A fila está vazia, nada a fazer
        finally:
            # Agenda a próxima verificação da fila
            self.root.after(100, self.process_ui_queue)

    def safe_log_message(self, message, level="INFO"):
        """Versão thread-safe de log_message que usa a fila de UI"""
        # Assegura que a atualização da UI ocorra na thread principal
        if threading.current_thread() is threading.main_thread():
            # Se estamos na thread principal, podemos atualizar diretamente
            self.log_message(message, level)
        else:
            # Se estamos em uma thread secundária, enfileiramos a atualização
            self.ui_queue.put(lambda: self.log_message(message, level))

    def safe_update_ui(self, widget_update_func):
        """Função genérica para atualizar widgets de forma thread-safe"""
        if threading.current_thread() is threading.main_thread():
            widget_update_func()
        else:
            self.ui_queue.put(widget_update_func)

    def log_message(self, message, level="INFO"):
        """Atualiza o widget de log (deve ser chamado apenas da thread principal)"""
        self.log_text.config(state="normal")
        tag = level.lower() if level.lower() in ["error", "success", "info", "warning", "command"] else "info"
        self.log_text.insert(tk.END, f"{message}\n", tag)
        self.log_text.config(state="disabled")
        self.log_text.see(tk.END) # Rola para a última mensagem

    def show_progress(self, show=True, message="", start_animation=True):

            if show:

    def load_api_keys(self):
        self.api_keys = []
        api_dir = "API_keys"
        api_file_path = os.path.join(api_dir, "api.txt")
        try:
            if not os.path.exists(api_dir):
                os.makedirs(api_dir)
                self.safe_log_message(f"Diretório {api_dir} criado.", "INFO")
            
            if os.path.exists(api_file_path):
                with open(api_file_path, "r", encoding="utf-8") as f:
                    keys = [line.strip() for line in f if line.strip()] # Lê uma chave por linha
                    if keys:
                        self.api_keys = keys
                        self.current_api_key_index = 0 # Começa com a primeira chave
                        self.safe_log_message(f"{len(self.api_keys)} chave(s) API carregada(s) de {api_file_path}.", "SUCCESS")
                    else:
                        self.safe_log_message(f"Arquivo {api_file_path} encontrado, mas está vazio ou não contém chaves válidas.", "WARNING")
            else:
                self.safe_log_message(f"Arquivo {api_file_path} não encontrado. Crie-o com suas chaves API (uma por linha).", "WARNING")
                # Cria um arquivo api.txt de exemplo se não existir
                with open(api_file_path, "w", encoding="utf-8") as f:
                    f.write("# Coloque suas chaves de API aqui, uma por linha\n")
                self.safe_log_message(f"Arquivo de exemplo {api_file_path} criado.", "INFO")

        except Exception as e:
            self.safe_log_message(f"Erro ao carregar chaves API: {e}", "ERROR")
            Messagebox.show_error("Erro de Chave API", f"Não foi possível carregar as chaves de API: {e}")

    def get_current_api_key(self):
        if not self.api_keys:
            # Tenta carregar as chaves novamente se estiverem vazias
            self.load_api_keys()
            if not self.api_keys: # Se ainda estiverem vazias após tentar recarregar
                self.safe_log_message("Nenhuma chave API disponível. Por favor, adicione chaves em API_keys/api.txt.", "ERROR")
                # Messagebox.show_error("Erro de API", "Nenhuma chave API configurada. Verifique API_keys/api.txt")
                return None 
        
        key = self.api_keys[self.current_api_key_index]
        # Rotaciona para a próxima chave para a próxima chamada (round-robin simples)
        self.current_api_key_index = (self.current_api_key_index + 1) % len(self.api_keys)
        return key

    def call_ai_api_for_edit(self, instruction, current_code, selected_lines_context=""):
        """Chama a API da OpenAI (ou outra configurada) para edição de código."""
        api_key = self.get_current_api_key()
        if not api_key:
            return {"success": False, "error": "Chave API não disponível."}

        endpoint = self.config.get("ai_edit_api_endpoint", "https://api.openai.com/v1/chat/completions")
        model = self.config.get("ai_edit_model", "gpt-3.5-turbo") # Modelo padrão para chat

        headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        }
        
        # Prepara o prompt para o modelo de chat
        system_message = "Você é um assistente de programação especializado em Cypress e JavaScript. Modifique o código fornecido estritamente de acordo com a instrução do usuário. Retorne APENAS o bloco de código modificado, sem explicações adicionais, comentários desnecessários ou markdown. Se a instrução não for clara ou não puder ser aplicada, retorne o código original sem modificações."
        
        user_prompt = f"Instrução: {instruction}\n"
        if selected_lines_context:
            user_prompt += f"\nContexto das linhas selecionadas:\n```javascript\n{selected_lines_context}```\n"
        user_prompt += f"\nCódigo atual a ser modificado:\n```javascript\n{current_code}```"
        
        payload = {
            "model": model,
            "messages": [
                {"role": "system", "content": system_message},
                {"role": "user", "content": user_prompt}
            ],
            "temperature": 0.2, # Baixa temperatura para manter o código mais determinístico
            "max_tokens": 2048 # Ajustar conforme necessário
        }

        try:
            self.safe_log_message(f"Enviando solicitação para IA (modelo: {model})...", "INFO")
            response = requests.post(endpoint, headers=headers, json=payload, timeout=60) # Timeout de 60 segundos
            response.raise_for_status() # Lança exceção para erros HTTP (4xx ou 5xx)
            
            response_data = response.json()
            
            if response_data.get("choices") and response_data["choices"][0].get("message") and response_data["choices"][0]["message"].get("content"):
                edited_code_raw = response_data["choices"][0]["message"]["content"].strip()
                
                # Tenta extrair apenas o bloco de código JavaScript
                match = re.search(r"```javascript\n(.*?)\n```", edited_code_raw, re.DOTALL)
                if match:
                    edited_code = match.group(1).strip()
                else:
                    # Se não encontrar o bloco delimitado, assume que a resposta inteira é o código (com fallback)
                    edited_code = edited_code_raw.replace("```javascript", "").replace("```", "").strip()
                
                self.safe_log_message("Resposta da IA recebida e processada.", "SUCCESS")
                return {"success": True, "edited_code": edited_code, "full_response": edited_code_raw}
            else:
                error_msg = "Resposta da API de IA não contém o conteúdo esperado."
                self.safe_log_message(f"{error_msg} Detalhes: {response_data}", "ERROR")
                return {"success": False, "error": error_msg, "details": response_data}
                
        except requests.exceptions.Timeout:
            self.safe_log_message("Timeout ao chamar a API de IA.", "ERROR")
            return {"success": False, "error": "A API de IA demorou muito para responder."}
        except requests.exceptions.RequestException as e:
            error_detail = str(e)
            if e.response is not None:
                try:
                    error_detail = e.response.json() # Tenta pegar detalhes do JSON de erro da API
                except json.JSONDecodeError:
                    error_detail = e.response.text # Se não for JSON, pega o texto bruto
            self.safe_log_message(f"Erro ao chamar API de IA: {e}. Detalhes: {error_detail}", "ERROR")
            return {"success": False, "error": f"Erro na comunicação com a API de IA: {e}", "details": error_detail}
        except Exception as e:
            self.safe_log_message(f"Erro inesperado ao processar resposta da IA: {e}", "ERROR")
            return {"success": False, "error": f"Erro inesperado: {e}"}

    def start_mapping(self):
        url_with_protocol = self.protocol_var.get() + self.url_entry.get()
        flow_name = self.flow_entry.get()
        api_key = self.api_key_entry.get()

        if not url_with_protocol or not flow_name:
            Messagebox.show_error("Erro", "URL e Nome do Fluxo são obrigatórios.")
            return

        if not self.api_keys and not api_key: # Verifica se não há chaves carregadas E o campo está vazio
             Messagebox.show_error("Erro", "Chave da API é obrigatória. Adicione em API_keys/api.txt ou no campo.")
             return
        
        # Se uma chave foi digitada no campo e não há chaves carregadas, usa a do campo e tenta salvar
        if api_key and api_key != "SUA_CHAVE_API_AQUI" and api_key not in self.api_keys:
            if not self.api_keys: # Se a lista de chaves carregadas estiver vazia
                self.api_keys.append(api_key)
                self.current_api_key_index = 0
                self.save_api_key(api_key) # Salva a nova chave no arquivo
            # Se já existem chaves carregadas, a chave do campo é apenas para esta sessão, não salva automaticamente
            # a menos que seja a única forma de ter uma chave.

        self.actions_log = []
        self.cypress_code = None
        self.log_text.config(state="normal")
        self.log_text.delete("1.0", tk.END)
        self.log_text.config(state="disabled")
        self.generate_button.config(state="disabled")
        self.save_button.config(state="disabled")
        self.start_button.grid_remove() # Oculta o botão Iniciar
        self.stop_button.grid()      # Mostra o botão Parar
        self.capturing = True
        self.stop_capture_event.clear()

        self.show_progress(message="Iniciando navegador...")
        self.safe_log_message(f"Iniciando mapeamento para: {url_with_protocol}", "INFO")

        # Inicia o navegador em uma nova thread para não bloquear a UI
        threading.Thread(target=self._start_browser_and_capture, args=(url_with_protocol,), daemon=True).start()

    def _start_browser_and_capture(self, url):
        try:
            # Tenta detectar o Chrome
            try:
                chrome_options = ChromeOptions()
                # chrome_options.add_argument("--headless") # Opcional: rodar em modo headless
                chrome_options.add_argument("--start-maximized")
                chrome_options.add_experimental_option("excludeSwitches", ["enable-automation"])
                chrome_options.add_experimental_option('useAutomationExtension', False)
                
                # Adiciona DevToolsActivePort para tentar capturar tráfego de rede (experimental)
                # chrome_options.add_argument("--remote-debugging-port=9222") 
                
                # Tenta encontrar o ChromeDriver automaticamente (Selenium Manager)
                self.driver = webdriver.Chrome(options=chrome_options)
                self.safe_log_message("Navegador Chrome iniciado.", "SUCCESS")
            except (SessionNotCreatedException, WebDriverException) as e_chrome:
                self.safe_log_message(f"Falha ao iniciar Chrome: {e_chrome}. Tentando Edge...", "WARNING")
                # Tenta detectar o Edge se o Chrome falhar
                try:
                    edge_options = EdgeOptions()
                    # edge_options.add_argument("--headless")
                    edge_options.add_argument("--start-maximized")
                    edge_options.add_experimental_option("excludeSwitches", ["enable-automation"])
                    edge_options.add_experimental_option('useAutomationExtension', False)
                    
                    # edge_options.add_argument("--remote-debugging-port=9223")

                    self.driver = webdriver.Edge(options=edge_options)
                    self.safe_log_message("Navegador Edge iniciado.", "SUCCESS")
                except (SessionNotCreatedException, WebDriverException) as e_edge:
                    self.safe_log_message(f"Falha ao iniciar Edge: {e_edge}. Verifique os drivers.", "ERROR")
                    self.show_progress(show=False)
                    self.safe_update_ui(lambda: self.stop_button.grid_remove())
                    self.safe_update_ui(lambda: self.start_button.grid()) # Mostra o botão Iniciar novamente
                    self.capturing = False
                    Messagebox.show_error("Erro de Navegador", "Não foi possível iniciar o Chrome ou Edge. Verifique se os navegadores e seus respectivos drivers (ChromeDriver/MSWebDriver) estão instalados e no PATH do sistema.")
                    return

            self.driver.get(url)
            self.show_progress(show=False) # Esconde a barra de progresso após o navegador abrir
            self.safe_log_message("Navegador aberto. Aguardando ações do usuário...", "INFO")
            self.safe_log_message("Pressione 'ESC' na janela do navegador para parar o mapeamento.", "INFO")

            # Inicia a captura de eventos do teclado em uma thread separada
            # keyboard.on_press_key("esc", self.stop_mapping_on_esc, suppress=True)
            # Substituído pelo botão "Parar Mapeamento"

            # Loop de captura de ações (simplificado, precisa de implementação robusta)
            # Esta parte é complexa e requer listeners de eventos no navegador
            # ou polling do DOM, que pode ser pesado.
            # A abordagem com Selenium puro para gravar ações do usuário é limitada.
            # Idealmente, usaríamos uma extensão de navegador ou proxy.
            
            # Exemplo de como capturar cliques e inputs (muito básico):
            last_url = self.driver.current_url
            while self.capturing and not self.stop_capture_event.is_set():
                try:
                    # Verifica se a janela do navegador ainda está aberta
                    if not self.driver.window_handles:
                        self.safe_log_message("Janela do navegador foi fechada. Parando mapeamento.", "WARNING")
                        self.stop_mapping() # Chama o método de parada
                        break

                    current_url = self.driver.current_url
                    if current_url != last_url:
                        self.actions_log.append({"action": "navigate", "url": current_url, "timestamp": time.time()})
                        self.safe_log_message(f"Navegou para: {current_url}", "INFO")
                        last_url = current_url
                    
                    # Lógica de captura de eventos (ex: JavaScript injetado para escutar cliques)
                    # Esta é a parte mais desafiadora e que pode causar lentidão se não otimizada.
                    # Por agora, vamos focar no botão de parada e na geração manual.
                    
                    # Captura de tráfego de rede (se habilitado)
                    if self.config.get("capture_network", False):
                        self._capture_network_traffic() # Implementar esta função

                except WebDriverException as e:
                    if "target window is closed" in str(e).lower() or \
                       "no such window" in str(e).lower() or \
                       "session deleted because of page crash" in str(e).lower() or \
                       "unable to connect to renderer" in str(e).lower():
                        self.safe_log_message(f"Navegador fechado ou crashou: {e}. Parando mapeamento.", "ERROR")
                        self.stop_mapping() # Chama o método de parada
                        break
                    else:
                        self.safe_log_message(f"Erro durante o mapeamento: {e}", "ERROR")
                        # Considerar parar o mapeamento dependendo do erro
                except Exception as e_loop:
                    self.safe_log_message(f"Erro inesperado no loop de captura: {e_loop}", "ERROR")
                
                time.sleep(0.2) # Pequena pausa para não sobrecarregar a CPU

        except TimeoutException:
            self.safe_log_message("Timeout ao carregar a URL inicial.", "ERROR")
            self.show_progress(show=False)
            Messagebox.show_error("Erro de Timeout", "A página demorou muito para carregar.")
        except Exception as e:
            self.safe_log_message(f"Erro ao iniciar o navegador: {e}", "ERROR")
            self.show_progress(show=False)
            Messagebox.show_error("Erro Crítico", f"Ocorreu um erro inesperado: {e}")
        finally:
            # Garante que o estado de captura seja resetado
            if self.capturing: # Se ainda estiver capturando, significa que não foi parado pelo stop_mapping
                self.safe_update_ui(lambda: self.stop_button.grid_remove())
                self.safe_update_ui(lambda: self.start_button.grid())
                self.capturing = False
            if self.driver:
                try:
                    self.driver.quit()
                    self.safe_log_message("Navegador fechado.", "INFO")
                except Exception as eq:
                    self.safe_log_message(f"Erro ao fechar navegador: {eq}", "WARNING")
                self.driver = None

    def _capture_network_traffic(self):
        """Captura tráfego de rede usando Selenium (limitado, idealmente via proxy/DevTools)"""
        # Esta é uma implementação MUITO SIMPLIFICADA e pode não funcionar bem ou ser pesada.
        # A API DevTools do Chrome/Edge é mais robusta para isso.
        try:
            if self.driver and hasattr(self.driver, "get_log"):
                for entry in self.driver.get_log('performance'): # 'performance' log pode conter eventos de rede
                    log = json.loads(entry['message'])['message']
                    if 'Network.requestWillBeSent' in log['method']:
                        request_id = log['params']['requestId']
                        pending_requests[request_id] = log['params']['request']
                        # self.safe_log_message(f"Network Request Out: {log['params']['request']['method']} {log['params']['request']['url'][:80]}...", "INFO")
                    elif 'Network.responseReceived' in log['method']:
                        request_id = log['params']['requestId']
                        if request_id in pending_requests:
                            request_info = pending_requests.pop(request_id)
                            response_info = log['params']['response']
                            network_requests.append({
                                "request": request_info,
                                "response": response_info,
                                "timestamp": time.time()
                            })
                            # self.safe_log_message(f"Network Response In: {response_info['status']} {request_info['url'][:80]}...", "INFO")
        except Exception as e:
            # self.safe_log_message(f"Erro ao capturar tráfego de rede: {e}", "WARNING")
            pass # Evita spammar o log com erros de captura de rede

    def get_best_selector(self, element_info):
        """Seleciona o melhor seletor com base na prioridade configurada."""
        if not element_info: return "body" # Fallback
        
        for selector_type in self.config.get("selector_priority", []):
            if self.config["selector_enabled"].get(selector_type, False):
                if selector_type == "data-cy" and element_info.get("data-cy"):
                    return f"[data-cy='{element_info['data-cy']}']"
                if selector_type == "data-testid" and element_info.get("data-testid"):
                    return f"[data-testid='{element_info['data-testid']}']"
                if selector_type == "id" and element_info.get("id"):
                    return f"#{element_info['id']}"
                if selector_type == "name" and element_info.get("name"):
                    return f"[name='{element_info['name']}']"
                # Adicionar mais tipos de seletores aqui (role, aria-label, class, tag, type)
                # Para 'class', pode ser necessário processar e escolher a classe mais específica.
                # Para 'tag' e 'type', pode ser menos robusto.

        return element_info.get("xpath", "body") # Fallback para XPath ou body

    def on_code_change(self, event, code_widget, line_numbers_widget):
        self.update_line_numbers(code_widget, line_numbers_widget)
        # Marcar que o código foi modificado para o histórico de undo/redo
        # A lógica de `update_code_history` deve ser chamada quando uma edição significativa é feita.
        # Aqui, apenas atualizamos os números de linha.

    def on_code_scroll(self, event_or_args, code_widget, line_numbers_widget):
        """Sincroniza o scroll do widget de números de linha com o widget de código."""
        if isinstance(event_or_args, tk.Event): # Evento do mouse wheel
            if event_or_args.delta < 0: # Rolar para baixo
                code_widget.yview_scroll(1, "units")
                line_numbers_widget.yview_scroll(1, "units")
            elif event_or_args.delta > 0: # Rolar para cima
                code_widget.yview_scroll(-1, "units")
                line_numbers_widget.yview_scroll(-1, "units")
        else: # Argumentos do scrollbar
            # O primeiro argumento é 'moveto' ou 'scroll'
            # O segundo é a fração (para moveto) ou número de unidades (para scroll)
            action, value = event_or_args[0], event_or_args[1]
            if action == "moveto":
                code_widget.yview_moveto(value)
                line_numbers_widget.yview_moveto(value)
            elif action == "scroll":
                code_widget.yview_scroll(int(value), event_or_args[2]) # event_or_args[2] é "units" ou "pages"
                line_numbers_widget.yview_scroll(int(value), event_or_args[2])
        self.update_line_numbers(code_widget, line_numbers_widget) # Garante que os números de linha estejam corretos após o scroll
        return "break" # Impede o processamento padrão do evento de scroll para evitar duplicação

    def update_line_numbers(self, code_widget, line_numbers_widget):
        line_numbers_widget.config(state="normal")
        line_numbers_widget.delete("1.0", tk.END)
        # Obtém o número de linhas visíveis e a primeira linha visível
        num_lines = code_widget.index(f"@0,{code_widget.winfo_height()}").split('.')[0] # Número de linhas que cabem na altura visível
        first_visible_line = code_widget.index("@0,0").split('.')[0]
        
        # Obtém o número total de linhas no widget de código
        total_lines_in_code = int(code_widget.index("end-1c").split('.')[0])

        # Gera os números de linha
        line_numbers_string = ""
        for i in range(1, total_lines_in_code + 1):
            line_numbers_string += str(i) + "\n"
        
        line_numbers_widget.insert("1.0", line_numbers_string)
        line_numbers_widget.config(state="disabled")
        
        # Sincroniza a visão vertical
        # Esta é a parte crucial para manter os números de linha alinhados com o texto
        # Precisamos garantir que a primeira linha visível no `code_widget` corresponda
        # à primeira linha visível no `line_numbers_widget`.
        # A maneira mais simples é forçar o yview do line_numbers_widget para o mesmo que o code_widget.
        # No entanto, o scroll já está sendo sincronizado pelo `on_code_scroll`.
        # Apenas garantir que o conteúdo é atualizado é suficiente aqui.
        # A sincronização de scroll é mais complexa do que parece devido a alturas de linha variáveis e fontes.
        # Para fontes monoespaçadas e alturas de linha fixas, a sincronização direta do yview_scroll funciona.
        current_scroll_pos = code_widget.yview()
        line_numbers_widget.yview_moveto(current_scroll_pos[0])

    def get_selected_lines_context(self, code_widget):
        """Obtém o texto das linhas que têm marcadores (checkboxes) selecionados."""
        # Esta função precisaria de uma lógica para checkboxes ao lado dos números de linha.
        # Por simplicidade, vamos assumir que se houver seleção de texto, esse é o contexto.
        try:
            selected_text = code_widget.get(tk.SEL_FIRST, tk.SEL_LAST)
            return selected_text.strip()
        except tk.TclError:
            return "" # Nenhuma seleção

    def _process_ai_request(self, instruction, current_code, selected_context, code_widget):
        ai_response = self.call_ai_api_for_edit(instruction, current_code, selected_context)
        self.show_progress(show=False)

        if ai_response.get("success"):
            self.current_ai_suggestion = ai_response.get("edited_code")
            full_response_display = ai_response.get("full_response", self.current_ai_suggestion) # Mostra a resposta completa se disponível
            
            self.add_chat_message(f"IA: Recebi sua solicitação. Aqui está uma sugestão de alteração:\n---\n{full_response_display}\n---", "ia")
            # Mostra os botões para aplicar ou rejeitar
            self.safe_update_ui(lambda: self.ai_suggestion_frame.pack(fill=tk.X, pady=5))
        else:
            error_message = ai_response.get("error", "Erro desconhecido da IA.")
            details = ai_response.get("details")
            display_error = f"IA: Desculpe, não consegui processar sua solicitação. Erro: {error_message}"
            if details:
                display_error += f"\nDetalhes: {str(details)[:200]}..." # Limita o tamanho dos detalhes
            self.add_chat_message(display_error, "error_ia")

    def update_code_history(self, code_content):
        # Se o índice não é o último, descarta o histórico futuro (redo)
        if self.history_index < len(self.code_history) - 1:
            self.code_history = self.code_history[:self.history_index + 1]
        
        # Evita adicionar estados duplicados consecutivos
        if not self.code_history or self.code_history[-1] != code_content:
            self.code_history.append(code_content)
            self.history_index = len(self.code_history) - 1
        
        # Limita o tamanho do histórico (opcional)
        # max_history_size = 50 
        # if len(self.code_history) > max_history_size:
        #     self.code_history.pop(0)
        #     self.history_index -=1

    def undo_edit(self, code_widget):
        if self.history_index > 0:
            self.history_index -= 1
            current_code_state = self.code_history[self.history_index]
            code_widget.config(state="normal")
            code_widget.delete("1.0", tk.END)
            code_widget.insert("1.0", current_code_state)
            code_widget.config(state="disabled")
            self.update_line_numbers(code_widget, self.line_numbers_text)
            self.add_chat_message("Ação Desfeita.", "system")

    def redo_edit(self, code_widget):
        if self.history_index < len(self.code_history) - 1:
            self.history_index += 1
            current_code_state = self.code_history[self.history_index]
            code_widget.config(state="normal")
            code_widget.delete("1.0", tk.END)
            code_widget.insert("1.0", current_code_state)
            code_widget.config(state="disabled")
            self.update_line_numbers(code_widget, self.line_numbers_text)
            self.add_chat_message("Ação Refeita.", "system")

    def save_api_key(self, api_key_to_save):
        """Adiciona uma nova chave API ao arquivo API_keys/api.txt."""
        api_dir = "API_keys"
        api_file_path = os.path.join(api_dir, "api.txt")
        try:
            if not os.path.exists(api_dir):
                os.makedirs(api_dir)
            
            # Lê chaves existentes para evitar duplicatas (opcional, mas bom)
            existing_keys = set()
            if os.path.exists(api_file_path):
                with open(api_file_path, "r", encoding="utf-8") as f_read:
                    for line in f_read:
                        existing_keys.add(line.strip())
            
            if api_key_to_save not in existing_keys:
                with open(api_file_path, "a", encoding="utf-8") as f_append:
                    f_append.write(f"{api_key_to_save}\n")
                self.safe_log_message(f"Nova chave API salva em {api_file_path}.", "SUCCESS")
                if api_key_to_save not in self.api_keys: # Adiciona à lista em memória também
                    self.api_keys.append(api_key_to_save)
            else:
                self.safe_log_message(f"Chave API já existe em {api_file_path}.", "INFO")

        except Exception as e:
            self.safe_log_message(f"Erro ao salvar nova chave API: {e}", "ERROR")

    def run_cypress_test(self, test_filepath, code_widget_ref, window_ref):
        """Executa o teste Cypress especificado, instalando dependências se necessário."""
        if not test_filepath:
            # Se não há um arquivo salvo, salva o conteúdo atual do editor em um temporário
            current_code_to_run = code_widget_ref.get("1.0", tk.END).strip()
            temp_dir = "temp_cypress_tests"
            os.makedirs(temp_dir, exist_ok=True)
            temp_filename = f"temp_test_{int(time.time())}.cy.js"
            test_filepath = os.path.join(temp_dir, temp_filename)
            try:
                with open(test_filepath, "w", encoding="utf-8") as f_temp:
                    f_temp.write(current_code_to_run)
                self.add_chat_message(f"Código atual salvo temporariamente em {test_filepath} para execução.", "system")
            except Exception as e_temp_save:
                self.add_chat_message(f"Erro ao salvar arquivo temporário: {e_temp_save}", "error_ia")
                Messagebox.show_error("Erro", f"Não foi possível salvar arquivo temporário para execução: {e_temp_save}", parent=window_ref)
                return
        elif os.path.exists(test_filepath):
            # Se o arquivo existe, atualiza-o com o conteúdo do editor antes de rodar
            current_code_to_run = code_widget_ref.get("1.0", tk.END).strip()
            try:
                with open(test_filepath, "w", encoding="utf-8") as f_update:
                    f_update.write(current_code_to_run)
                self.add_chat_message(f"Arquivo {test_filepath} atualizado com o código do editor.", "system")
            except Exception as e_update_save:
                self.add_chat_message(f"Erro ao atualizar {test_filepath}: {e_update_save}", "error_ia")
                Messagebox.show_error("Erro", f"Não foi possível atualizar {test_filepath}: {e_update_save}", parent=window_ref)
                return
        else:
            Messagebox.show_error("Erro", f"Arquivo de teste não encontrado: {test_filepath}", parent=window_ref)
            self.add_chat_message(f"Arquivo de teste não encontrado: {test_filepath}", "error_ia")
            return

        project_dir = os.path.dirname(os.path.dirname(os.path.abspath(test_filepath))) # Assume que o teste está em 'project_dir/cypress/e2e/teste.cy.js'
        # Se o arquivo temporário está em 'temp_cypress_tests', o project_dir será o diretório pai de 'temp_cypress_tests'
        if "temp_cypress_tests" in test_filepath:
            project_dir = os.path.abspath(os.path.join(os.path.dirname(test_filepath), "..")) # Diretório pai de temp_cypress_tests
            # Cria uma estrutura mínima de projeto Cypress se não existir
            os.makedirs(os.path.join(project_dir, "cypress", "e2e"), exist_ok=True)
            # Copia o teste temporário para a estrutura esperada pelo Cypress
            target_cypress_e2e_dir = os.path.join(project_dir, "cypress", "e2e")
            final_test_path_in_project = os.path.join(target_cypress_e2e_dir, os.path.basename(test_filepath))
            shutil.copy2(test_filepath, final_test_path_in_project)
            test_filepath_for_run_command = final_test_path_in_project # Usar este caminho para o comando run
        else:
            test_filepath_for_run_command = os.path.abspath(test_filepath)

        self.add_chat_message(f"Preparando para executar: {os.path.basename(test_filepath_for_run_command)} em {project_dir}", "system")
        self.show_progress(message=f"Executando {os.path.basename(test_filepath_for_run_command)}...", start_animation=True)

        # Executa em uma thread para não bloquear a UI
        threading.Thread(target=self._execute_cypress_thread, args=(project_dir, test_filepath_for_run_command, window_ref), daemon=True).start()

    def _execute_cypress_thread(self, project_dir, absolute_test_filepath, parent_window):
        log_widget = self.ai_chat_log # Usar o log do chat IA para feedback da execução
        original_state = log_widget.cget("state")
        self.safe_update_ui(lambda: log_widget.config(state="normal"))

        try:
            # 1. Verificar Node.js
            self.safe_update_ui(lambda: self.add_chat_message("Verificando Node.js...", "system"))
            node_path = shutil.which("node")
            npm_path = shutil.which("npm")
            npx_path = shutil.which("npx")
            nodeenv_path = shutil.which("nodeenv")

            if not (node_path and npm_path and npx_path):
                self.safe_update_ui(lambda: self.add_chat_message("Node.js, npm ou npx não encontrados globalmente.", "warning"))
                if nodeenv_path:
                    self.safe_update_ui(lambda: self.add_chat_message("Tentando criar ambiente Node.js local com nodeenv...", "system"))
                    venv_dir = os.path.join(project_dir, ".nodeenv") # Ambiente Node.js local
                    if not os.path.exists(os.path.join(venv_dir, "bin", "node")):
                        nodeenv_create_res = subprocess.run([nodeenv_path, venv_dir, "--node=lts", "--force"], capture_output=True, text=True, cwd=project_dir)
                        if nodeenv_create_res.returncode != 0:
                            raise Exception(f"Erro ao criar ambiente nodeenv: {nodeenv_create_res.stderr}")
                        self.safe_update_ui(lambda: self.add_chat_message("Ambiente nodeenv LTS criado.", "success"))
                    else:
                        self.safe_update_ui(lambda: self.add_chat_message("Usando ambiente nodeenv existente.", "info"))
                    
                    # Atualiza os caminhos para usar o nodeenv
                    activate_script = "activate.bat" if os.name == 'nt' else "activate"
                    node_path = os.path.join(venv_dir, "Scripts" if os.name == 'nt' else "bin", "node")
                    npm_path = os.path.join(venv_dir, "Scripts" if os.name == 'nt' else "bin", "npm")
                    npx_path = os.path.join(venv_dir, "Scripts" if os.name == 'nt' else "bin", "npx")
                    if not (os.path.exists(node_path) and os.path.exists(npm_path) and os.path.exists(npx_path)):
                         raise Exception("Falha ao configurar caminhos do nodeenv.")
                else:
                    raise Exception("Node.js/npm/npx não encontrado e nodeenv não está disponível. Instale Node.js ou nodeenv (pip install nodeenv).")
            
            self.safe_update_ui(lambda: self.add_chat_message(f"Node.js encontrado: {node_path}", "success"))
            npm_cmd = [npm_path]
            npx_cmd = [npx_path]

            # 2. Verificar package.json e criar se não existir
            package_json_path = os.path.join(project_dir, "package.json")
            if not os.path.exists(package_json_path):
                self.safe_update_ui(lambda: self.add_chat_message("package.json não encontrado. Criando...", "system"))
                init_res = subprocess.run(npm_cmd + ["init", "-y"], cwd=project_dir, capture_output=True, text=True, shell=(os.name=='nt'))
                if init_res.returncode != 0:
                    raise Exception(f"Erro ao criar package.json: {init_res.stderr}")
                self.safe_update_ui(lambda: self.add_chat_message("package.json criado.", "success"))

            # 3. Verificar Cypress
            self.safe_update_ui(lambda: self.add_chat_message("Verificando Cypress...", "system"))
            cypress_installed = False
            try:
                # Verifica se o Cypress está listado em dependências ou se o comando npx cypress --version funciona
                with open(package_json_path, "r") as f_pkg:
                    pkg_data = json.load(f_pkg)
                    if "cypress" in pkg_data.get("devDependencies", {}) or "cypress" in pkg_data.get("dependencies", {}):
                        cypress_installed = True
                if not cypress_installed:
                    cypress_version_res = subprocess.run(npx_cmd + ["cypress", "--version"], cwd=project_dir, capture_output=True, text=True, shell=(os.name=='nt'), timeout=30)
                    if cypress_version_res.returncode == 0 and "Cypress" in cypress_version_res.stdout:
                        cypress_installed = True
                        self.safe_update_ui(lambda: self.add_chat_message(f"Cypress encontrado: {cypress_version_res.stdout.strip()}", "success"))
            except (subprocess.TimeoutExpired, FileNotFoundError, Exception):
                cypress_installed = False # Assume não instalado se a verificação falhar
            
            if not cypress_installed:
                self.safe_update_ui(lambda: self.add_chat_message("Cypress não encontrado. Tentando instalar...", "warning"))
                install_cypress_q = queue.Queue()
                def ask_install_dialog():
                    if Messagebox.yesno("Cypress não encontrado. Deseja instalá-lo neste projeto?", "Instalar Cypress?", parent=parent_window) == "Yes":
                        install_cypress_q.put(True)
                    else:
                        install_cypress_q.put(False)
                
                # Executa o diálogo na thread da UI
                self.safe_update_ui(ask_install_dialog)
                # Espera pela resposta do diálogo (bloqueia esta thread, não a UI)
                should_install = install_cypress_q.get(timeout=60) 

                if should_install:
                    self.safe_update_ui(lambda: self.add_chat_message("Instalando Cypress (pode levar alguns minutos)...", "system"))
                    install_res = subprocess.run(npm_cmd + ["install", "cypress", "--save-dev"], cwd=project_dir, capture_output=True, text=True, shell=(os.name=='nt'))
                    if install_res.returncode != 0:
                        raise Exception(f"Erro ao instalar Cypress: {install_res.stderr}")
                    self.safe_update_ui(lambda: self.add_chat_message("Cypress instalado com sucesso.", "success"))
                else:
                    self.safe_update_ui(lambda: self.add_chat_message("Instalação do Cypress cancelada pelo usuário.", "info"))
                    self.show_progress(show=False)
                    self.safe_update_ui(lambda: log_widget.config(state=original_state))
                    return
            
            # 4. Garantir estrutura de pastas e cypress.config.js
            cypress_e2e_dir = os.path.join(project_dir, "cypress", "e2e")
            os.makedirs(cypress_e2e_dir, exist_ok=True)
            cypress_config_path = os.path.join(project_dir, "cypress.config.js")
            if not os.path.exists(cypress_config_path):
                self.safe_update_ui(lambda: self.add_chat_message("cypress.config.js não encontrado. Criando configuração padrão...", "system"))
                with open(cypress_config_path, "w", encoding="utf-8") as f_cfg:
                    f_cfg.write("const { defineConfig } = require(\"cypress\")\n\nmodule.exports = defineConfig({\n  e2e: {\n    setupNodeEvents(on, config) {\n      // implement node event listeners here\n    },\n    specPattern: \"cypress/e2e/**/*.cy.js\"\n  },\n});")
                self.safe_update_ui(lambda: self.add_chat_message("cypress.config.js criado.", "success"))

            # 5. Executar o teste
            # O absolute_test_filepath já deve estar dentro da estrutura cypress/e2e do project_dir
            relative_spec_path = os.path.relpath(absolute_test_filepath, project_dir).replace("\\", "/")
            self.safe_update_ui(lambda: self.add_chat_message(f"Executando teste: {relative_spec_path}...", "command"))
            
            run_res = subprocess.run(npx_cmd + ["cypress", "run", "--spec", relative_spec_path], cwd=project_dir, capture_output=True, text=True, shell=(os.name=='nt'))
            
            if run_res.stdout:
                self.safe_update_ui(lambda: self.add_chat_message(f"Saída do Cypress:\n{run_res.stdout}", "info"))
            if run_res.stderr:
                # Alguns logs do Cypress vão para stderr mesmo em sucesso, então não marcar como erro sempre
                log_level_stderr = "error_ia" if run_res.returncode != 0 else "warning"
                self.safe_update_ui(lambda: self.add_chat_message(f"Saída de Erro do Cypress:\n{run_res.stderr}", log_level_stderr))
            
            if run_res.returncode == 0:
                self.safe_update_ui(lambda: self.add_chat_message("Teste Cypress executado com sucesso!", "success"))
            else:
                self.safe_update_ui(lambda: self.add_chat_message(f"Teste Cypress falhou (código de saída: {run_res.returncode}). Verifique os logs acima.", "error_ia"))

        except Exception as e_run_thread:
            self.safe_update_ui(lambda: self.add_chat_message(f"Erro crítico durante a execução do teste: {e_run_thread}", "error_ia"))
            # Messagebox.show_error("Erro de Execução", f"Falha ao executar o teste: {e_run_thread}", parent=parent_window) # Pode ser muito intrusivo
        finally:
            self.show_progress(show=False)
            self.safe_update_ui(lambda: log_widget.config(state=original_state))
            # Limpa o arquivo temporário se foi criado
            if "temp_cypress_tests" in absolute_test_filepath and os.path.exists(absolute_test_filepath):
                try: 
                    # os.remove(absolute_test_filepath) # O arquivo foi copiado, o original pode ser removido
                    # Melhor remover o diretório temp_cypress_tests se estiver vazio ou apenas ele.
                    temp_dir_to_clean = os.path.dirname(absolute_test_filepath)
                    if os.path.basename(temp_dir_to_clean) == "e2e": # Se o temp foi copiado para e2e
                        # A limpeza do projeto temporário é mais complexa, pode ser melhor deixar para o usuário
                        pass 
                except Exception as e_clean:
                    self.safe_update_ui(lambda: self.add_chat_message(f"Aviso: não foi possível limpar arquivo temporário {absolute_test_filepath}: {e_clean}", "warning"))


if __name__ == "__main__":
