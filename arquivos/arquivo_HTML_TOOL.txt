import tkinter as tk
from tkinter import ttk, messagebox, simpledialog

def mostrar_painel_apis(janela, driver, historico, ajustar_janela, voltar_callback, capture_network_traffic, network_requests):
    """Exibe um painel com as requisições de API capturadas e permite visualizar respostas com token."""
    for widget in janela.winfo_children():
        widget.destroy()
    janela.title("Autenticação para Visualizar Requisições")

    # Tela de autenticação
    tk.Label(janela, text="Por favor, forneça as informações de autenticação necessárias:").pack(pady=5)

    # Token Bearer
    tk.Label(janela, text="Token Bearer (deixe em branco se não necessário):").pack(pady=5)
    token_entry = tk.Entry(janela, width=50)
    token_entry.pack(pady=5)

    # Cookies
    tk.Label(janela, text="Cookies (formato: nome=valor; nome=valor, deixe em branco se não necessário):").pack(pady=5)
    cookies_entry = tk.Entry(janela, width=50)
    cookies_entry.pack(pady=5)

    # Cabeçalhos personalizados
    tk.Label(janela, text="Cabeçalhos personalizados (formato: chave=valor; chave=valor, deixe em branco se não necessário):").pack(pady=5)
    headers_entry = tk.Entry(janela, width=50)
    headers_entry.pack(pady=5)

    def prosseguir():
        # Coleta as informações de autenticação
        token = token_entry.get().strip()
        cookies_raw = cookies_entry.get().strip()
        headers_raw = headers_entry.get().strip()

        # Processa cookies
        cookies = {}
        if cookies_raw:
            for cookie in cookies_raw.split(';'):
                if '=' in cookie:
                    key, value = cookie.split('=', 1)
                    cookies[key.strip()] = value.strip()

        # Processa cabeçalhos personalizados
        custom_headers = {}
        if headers_raw:
            for header in headers_raw.split(';'):
                if '=' in header:
                    key, value = header.split('=', 1)
                    custom_headers[key.strip()] = value.strip()

        # Prossegue para o painel de APIs
        exibir_painel(token, cookies, custom_headers)

    def exibir_painel(token, cookies, custom_headers):
        for widget in janela.winfo_children():
            widget.destroy()
        janela.title("Visualizador de APIs")
        historico.append("apis")

        tk.Label(janela, text="Requisições de API em Execução:").pack(pady=5)

        # Tabela com colunas para método e URL
        colunas = ("Method", "URL")
        tree = ttk.Treeview(janela, columns=colunas, show="headings", height=15)
        tree.heading("Method", text="Método")
        tree.heading("URL", text="URL da Requisição")
        tree.column("Method", width=100)
        tree.column("URL", width=500)
        tree.pack(fill="both", expand=True, padx=5, pady=5)

        atualizando = True
        after_id = None

        def atualizar_tabela():
            nonlocal after_id
            if not atualizando or not janela.winfo_exists():
                return
            
            # Captura as requisições mais recentes
            capture_network_traffic()
            
            urls_atuais = {tree.item(item)["values"][1] for item in tree.get_children() if tree.item(item)["values"]}
            
            for req in network_requests:
                url = req["url"]
                method = req["method"]
                if url not in urls_atuais:
                    tree.insert("", "end", values=(method, url))
                    print(f"Adicionando requisição ao painel: {method} {url}")  # Log para depuração
                    print(f"Request Headers: {req['requestHeaders']}")  # Log dos cabeçalhos enviados
                    print(f"Request Body: {req['requestBody']}")  # Log do corpo da requisição
                    if "responseHeaders" in req:
                        print(f"Response Headers: {req['responseHeaders']}")
                        print(f"Response Body: {req['responseBody']}")
            
            after_id = janela.after(1000, atualizar_tabela)

        def on_url_click(event):
            item = tree.selection()
            if not item:
                return
            method, url = tree.item(item[0], "values")

            headers = custom_headers.copy()
            if token:
                headers["Authorization"] = f"Bearer {token}"
            headers["Sistemaid"] = "1"
            headers["Content-Type"] = "application/json"
            headers["Accept"] = "application/json, text/plain, */*"

            try:
                if method == "GET":
                    response = requests.get(url, headers=headers, cookies=cookies, timeout=10)
                elif method == "POST":
                    response = requests.post(url, headers=headers, cookies=cookies, timeout=10)
                elif method == "PUT":
                    response = requests.put(url, headers=headers, cookies=cookies, timeout=10)
                elif method == "DELETE":
                    response = requests.delete(url, headers=headers, cookies=cookies, timeout=10)
                else:
                    messagebox.showerror("Erro", f"Método HTTP {method} não suportado.")
                    return

                response.raise_for_status()

                janela_resposta = tk.Toplevel(janela)
                janela_resposta.title(f"Resposta da API: {url}")
                janela_resposta.geometry("600x400")

                text_area = tk.Text(janela_resposta, wrap="word")
                text_area.insert("1.0", response.text)
                text_area.pack(fill="both", expand=True, padx=5, pady=5)

                tk.Button(janela_resposta, text="Fechar", command=janela_resposta.destroy).pack(pady=5)
            except requests.RequestException as e:
                messagebox.showerror("Erro", f"Falha ao acessar a API: {str(e)}")

        def parar_atualizacao():
            nonlocal atualizando, after_id
            atualizando = False
            if after_id is not None:
                janela.after_cancel(after_id)
                after_id = None

        tree.bind("<Double-1>", on_url_click)

        tk.Button(janela, text="Voltar", command=lambda: [parar_atualizacao(), voltar_callback()]).pack(pady=5)

        atualizar_tabela()
        ajustar_janela()

        janela.protocol("WM_DELETE_WINDOW", lambda: [parar_atualizacao(), janela.destroy()])

    tk.Button(janela, text="Prosseguir", command=prosseguir).pack(pady=10)
    ajustar_janela()

import os

def verificar_data_cy(elemento):
    """Verifica se o elemento tem o atributo [data-cy]."""
    return "data-cy" in elemento.attrs

def adicionar_indicador_data_cy(tree, elementos, checkboxes_globais):
    """Adiciona uma exclamação vermelha na coluna Categoria para elementos sem [data-cy]."""
    for item in tree.get_children():
        tree.delete(item)
    
    for elemento in elementos:
        id_ = elemento["id"]
        texto = elemento["texto"] if elemento["texto"] else "(Nenhum)"
        aria = elemento["aria_label"] if elemento["aria_label"] else "(Nenhum)"
        categoria = elemento["categoria"]
        tem_data_cy = verificar_data_cy(elemento["elemento"])
        
        if id_ not in checkboxes_globais:
            checkboxes_globais[id_] = {"var": tk.BooleanVar(value=False), "elemento": elemento["elemento"]}
        var = checkboxes_globais[id_]["var"]
        
        # Se não tem [data-cy], adiciona uma exclamação vermelha antes da categoria
        categoria_display = f"⚠ {categoria}" if not tem_data_cy else categoria
        
        item_id = tree.insert("", "end", values=("✔" if var.get() else "", id_, texto, aria, categoria_display))
        
        # Aplica a cor vermelha apenas ao símbolo ⚠, mantendo o texto em preto
        if not tem_data_cy:
            tree.tag_configure("sem_data_cy", foreground="red")
            # Aplica a tag apenas ao texto da célula "Categoria"
            tree.set(item_id, "Categoria", categoria_display)

def configurar_hover_data_cy(tree):
    """Configura o evento de hover para mostrar/sumir a mensagem 'elemento sem data-cy'."""
    tooltip = None
    
    def mostrar_mensagem(event):
        nonlocal tooltip
        # Identifica a célula sob o mouse
        item = tree.identify_row(event.y)
        if not item:
            return
        col = tree.identify_column(event.x)
        if col != "#5":  # Coluna "Categoria"
            return
        
        valores = tree.item(item, "values")
        categoria = valores[4]
        if not categoria.startswith("⚠"):
            return
        
        # Remove qualquer tooltip existente antes de criar um novo
        if tooltip:
            tooltip.destroy()
        
        # Cria um novo tooltip
        x, y = event.x_root, event.y_root
        tooltip = tk.Toplevel(tree)
        tooltip.wm_overrideredirect(True)
        tooltip.wm_geometry(f"+{x+10}+{y+10}")
        label = tk.Label(tooltip, text="elemento sem data-cy", bg="white", fg="black", relief="solid", borderwidth=1)
        label.pack()
    
    def esconder_mensagem(event):
        nonlocal tooltip
        # Remove o tooltip ao sair da célula
        if tooltip:
            tooltip.destroy()
            tooltip = None
    
    # Vincula os eventos
    tree.bind("<Enter>", lambda e: None)  # Necessário para evitar comportamento indesejado
    tree.bind("<Motion>", mostrar_mensagem)
    tree.bind("<Leave>", esconder_mensagem)

def denunciar_falta_data_cy(checkboxes_globais, pasta):
    """Gera um arquivo de texto com os elementos selecionados sem [data-cy]."""
    selecionados = {id_: info["elemento"] for id_, info in checkboxes_globais.items() if info["var"].get()}
    if not selecionados:
        messagebox.showwarning("Aviso", "Nenhum elemento selecionado para denunciar!")
        return
    
    elementos_sem_data_cy = [elem for id_, elem in selecionados.items() if not verificar_data_cy(elem)]
    if not elementos_sem_data_cy:
        messagebox.showinfo("Info", "Todos os elementos selecionados possuem [data-cy]!")
        return
    
    conteudo = ""
    for i, elemento in enumerate(elementos_sem_data_cy, 1):
        soup = BeautifulSoup(str(elemento), "html.parser")
        elem = soup.find()  # Pega o primeiro elemento da string BeautifulSoup
        categoria = determinar_categoria(elem)
        aria_label = elem.get("aria-label", "(Nenhum)")
        texto = elem.get_text(strip=True) or "(Nenhum)"
        id_elem = elem.get("id", None)
        
        if aria_label == "(Nenhum)" and texto == "(Nenhum)" and id_elem:
            identificador = f"ID = '{id_elem}'"
        else:
            identificador = aria_label if aria_label != "(Nenhum)" else texto
        
        conteudo += f"{i} - Elemento da categoria [{categoria}] sem data-cy, [{identificador}]\n"
    
    nome_arquivo = "denuncia_falta_data_cy"
    caminho_arquivo = os.path.join(pasta, f"{nome_arquivo}.txt")
    base, ext = os.path.splitext(caminho_arquivo)
    contador = 1
    while os.path.exists(caminho_arquivo):
        caminho_arquivo = f"{base}_{contador}{ext}"
        contador += 1
    
    with open(caminho_arquivo, "w", encoding="utf-8") as arquivo:
        arquivo.write(conteudo)
    
    messagebox.showinfo("Sucesso", f"Denúncia salva em: {caminho_arquivo}")

def determinar_categoria(elemento):
    """Determina a categoria do elemento."""
    tag = elemento.name.lower()
    attrs = elemento.attrs
    if tag == "button" or attrs.get("role", "").lower() == "button":
        return "Botão"
    elif tag == "input":
        tipo = attrs.get("type", "").lower()
        if tipo in ["button", "submit", "reset"]:
            return "Botão"
        elif tipo in ["text", "email", "password", "search", "tel", "url"]:
            return "Input"
    elif tag == "select":
        return "Select"
    elif tag == "a" and attrs.get("href"):
        return "Link"
    elif elemento.get_text(strip=True) and not attrs.get("role", "").lower() in ["combobox", "menuitem"]:
        return "Texto"
    elif attrs.get("role", "").lower() in ["combobox", "menuitem"]:
        return "Select"
    elif elemento.get("onclick") or elemento.get("tabindex") is not None:
        return "Outros"
    return "Outros"

def filtrar_html(html, filtro):
    soup = BeautifulSoup(html, "html.parser")
    resultado = ""
    if filtro == "HTML Completo":
        return remover_base64(html)
    elif filtro == "Apenas IDs de Botões":
        botoes = soup.find_all(["button", lambda tag: tag.name == "input" and tag.get("type") in ["button", "submit", "reset"]])
        botoes_customizados = soup.find_all(lambda tag: tag.get("role") == "button")
        resultado = "IDs de Botões encontrados:\n"
        ids_encontrados = set()
        for botao in botoes + botoes_customizados:
            id_botao = botao.get("id")
            if id_botao:
                ids_encontrados.add(f"ID: {id_botao} (Tag: {botao.name})")
        resultado += "\n".join(sorted(ids_encontrados))
        return resultado if ids_encontrados else "Nenhum ID de botão encontrado."
    elif filtro == "Apenas Inputs":
        inputs = soup.find_all("input")
        resultado = "Inputs encontrados:\n"
        for inp in inputs:
            resultado += str(inp) + "\n"
        return resultado if resultado != "Inputs encontrados:\n" else "Nenhum input encontrado."
    return "Filtro inválido."

def gerar_seletores(elemento):
    seletores = []
    tag = elemento.name
    atributos = elemento.attrs
    seletores.append(tag)
    if "id" in atributos:
        seletores.append(f"#{atributos['id']}")
        seletores.append(f"{tag}#{atributos['id']}")
    if "class" in atributos:
        classes = atributos["class"]
        for classe in classes:
            seletores.append(f".{classe}")
            seletores.append(f"{tag}.{classe}")
        if len(classes) > 1:
            seletores.append(f"{tag}.{'.'.join(classes)}")
    atributos_relevantes = ["aria-label", "aria-expanded", "aria-controls", "aria-owns", "role", "type", "data-cy", "tabindex", "name"]
    for attr in atributos_relevantes:
        if attr in atributos and atributos[attr]:
            valor = atributos[attr]
            if isinstance(valor, (list, tuple)):
                valor = valor[0]
            seletores.append(f'{tag}[{attr}="{valor}"]')
            if "id" in atributos:
                seletores.append(f'#{atributos["id"]}[{attr}="{valor}"]')
    attrs_combinados = []
    for attr in atributos_relevantes:
        if attr in atributos and atributos[attr]:
            valor = atributos[attr]
            if isinstance(valor, (list, tuple)):
                valor = valor[0]
            attrs_combinados.append(f'[{attr}="{valor}"]')
    if len(attrs_combinados) > 1:
        seletores.append(f"{tag}{''.join(attrs_combinados)}")
    return sorted(set(seletores))

import tkinter as tk
from tkinter import ttk, messagebox, simpledialog

visited_urls = []  # Global list to store visited URLs

def add_url(url):
    """Add a URL to the history."""
    visited_urls.append(url)

def mostrar_historico_urls(janela, historico, voltar, pasta_arquivos, ajustar_janela):
    """Display a panel with the history of visited URLs."""
    for widget in janela.winfo_children():
        widget.destroy()
    janela.title("Histórico de URLs")
    tk.Label(janela, text="Histórico de URLs visitadas:").pack(pady=5)

    colunas = ("Select", "URL")
    tree = ttk.Treeview(janela, columns=colunas, show="headings", height=15)
    tree.heading("Select", text="Selecionar")
    tree.heading("URL", text="URL")
    tree.column("Select", width=80, anchor="center")
    tree.column("URL", width=400)
    tree.pack(fill="both", expand=True, padx=5, pady=5)

    checkboxes_urls = {i: tk.BooleanVar(value=False) for i in range(len(visited_urls))}

    for i, url in enumerate(visited_urls):
        checkbox_text = "☐"
        tree.insert("", "end", iid=i, values=(checkbox_text, url))

    def toggle_checkbox(event):
        item = tree.identify_row(event.y)
        if item and tree.identify_column(event.x) == "#1":
            i = int(item)
            current_value = checkboxes_urls[i].get()
            checkboxes_urls[i].set(not current_value)
            checkbox_text = "☑" if not current_value else "☐"
            tree.item(item, values=(checkbox_text, visited_urls[i]))

    tree.bind("<Button-1>", toggle_checkbox)

    def exportar_urls():
        selected_indices = [i for i in range(len(visited_urls)) if checkboxes_urls[i].get()]
        if not selected_indices:
            messagebox.showwarning("Aviso", "Nenhuma URL selecionada para exportar!")
            return

        resposta = messagebox.askyesno("Exportar", "Deseja adicionar descrições às URLs selecionadas?")
        desc_dict = {}
        if resposta:
            desc_window = tk.Toplevel(janela)
            desc_window.title("Adicionar Descrições")
            descriptions = {i: tk.StringVar() for i in selected_indices}
            for i in selected_indices:
                tk.Label(desc_window, text=f"URL: {visited_urls[i]}").pack()
                tk.Entry(desc_window, textvariable=descriptions[i]).pack()
            tk.Button(desc_window, text="OK", command=desc_window.destroy).pack()
            desc_window.grab_set()
            janela.wait_window(desc_window)
            desc_dict = {i: descriptions[i].get() for i in selected_indices}
        else:
            desc_dict = {i: "" for i in selected_indices}

        conteudo = ""
        for idx, i in enumerate(selected_indices, start=1):
            url = visited_urls[i]
            desc = desc_dict[i]
            conteudo += f"URL{idx} {url}, descrição:{desc};\n\n"

        nome_arquivo = simpledialog.askstring("Exportar", "Digite o nome do arquivo (sem extensão):", initialvalue="historico_urls")
        if nome_arquivo:
            from htmlCatcher import salvar_html_em_txt  # Import here to avoid circular import
            caminho_arquivo = salvar_html_em_txt("Histórico de URLs", conteudo, pasta_arquivos, nome_arquivo)
            messagebox.showinfo("Sucesso", f"Histórico de URLs salvo em: {caminho_arquivo}")

    tk.Button(janela, text="Exportar .txt", command=exportar_urls).pack(pady=5)
    tk.Button(janela, text="Voltar", command=voltar).pack(pady=5)
    ajustar_janela()
